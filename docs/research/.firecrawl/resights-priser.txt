    // Find tab titles + contents i Elementor Tabs
    const titleNodes = Array.from(tabsEl.querySelectorAll('.elementor-tab-title'));
    const contentNodes = Array.from(tabsEl.querySelectorAll('.elementor-tab-content'));

    if (!titleNodes.length || !contentNodes.length) return;

    // Begræns til 4 (så vi ikke får “ekstra” dots)
    const count = Math.min(MAX_DOTS, titleNodes.length, contentNodes.length);

    // Find den hvide boks (kortet) og placer dots EFTER kortet (udenfor)
    const card = tabsEl.closest('.resights-slider-card');
    if (!card) return;

    const existing = card.parentElement?.querySelector(':scope > .resights-slider-dots');
    if (existing) existing.remove();

    dotsWrap.className = 'resights-slider-dots';

    const dots = [];
    for (let i = 0; i < count; i++) {
      const dot = document.createElement('button');
      dot.type = 'button';
      dot.className = 'resights-slider-dot';
      dot.setAttribute('aria-label', `Slide ${i + 1}`);
      dot.addEventListener('click', function () {
        activate(i, true);
      });
      dotsWrap.appendChild(dot);
      dots.push(dot);

      dots.forEach((d, i) => d.classList.toggle('is-active', i === idx));

      activeIndex = idx;

      if (userAction && AUTOPLAY) {
        stop();
        start();
      }
    }

    function start() {
      if (!AUTOPLAY) return;
      stop();
      timer = window.setInterval(() => {
        const next = (activeIndex + 1) % count;
        activate(next, false);
      }, INTERVAL_MS);
    }

    function stop() {
      if (timer) window.clearInterval(timer);
      timer = null;
    }

    // Sørg for at kun de første 4 “slides” er relevante
    // (vi klikker kun på de første 4)
    // Start på første tab
    activate(0, false);
    start();

    // Hvis Elementor/preview re-renderer, så prøv at holde dots synkron
    const observer = new MutationObserver(() => {
      // Find hvilket tab der er aktivt (Elementor sætter ofte .elementor-active)
      const activeTitleIndex = titleNodes.findIndex(n => n.classList.contains('elementor-active'));
      if (activeTitleIndex >= 0 && activeTitleIndex < count) {
        activeIndex = activeTitleIndex;
        setActiveDot(activeIndex);
      }
    });

    observer.observe(tabsEl, { subtree: true, attributes: true, attributeFilter: ['class'] });
  }

  function initAll() {
    const all = document.querySelectorAll('.resights-tabs');
    all.forEach(initOneSlider);
  }

  // Kør når DOM er klar
  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initAll);
  } else {
    initAll();
  }
